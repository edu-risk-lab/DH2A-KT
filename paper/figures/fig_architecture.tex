  passbox/.style={draw, rounded corners=2pt, align=center,
              inner sep=4pt, text width=4.05cm, minimum height=1.85cm,
              fill=green!8},
  warnbox/.style={draw, rounded corners=2pt, align=center,
              inner sep=4pt, text width=4.05cm, minimum height=1.85cm,

\node[box] (p0) {Train-only\\construction\\$E_{\mathrm{pre}}$ / Q--KC};
\node[box, right=0.32cm of p0] (audit) {Hyperedge\\membership audit};

\draw[->, thick] (fork) -| (fail.north);
\end{tikzpicture}
\caption{The credit protocol. Auxiliary knowledge is built from training
learners only and audited for membership leakage. Each message is then
placed on a rung with a designated control, run on five paired seeds,
and assigned one of three outcomes. The boxes give the outcomes of this
paper (Table~\ref{tab:credit-transfer}); the same time message is
credited or not depending on corpus, encoder, recipe, timestamp
export, and rung.}
